
# 安装依赖（本仓库安全规范：不执行生命周期脚本）
npm install --ignore-scripts

# 打包
npm run build          # 先从网络刷新模型数据（models.dev、OpenRouter 等），再构建
npm run build:offline  # 无网络环境：复用已有的模型数据构建

# 运行
node packages/coding-agent/dist/bundle/cli.js

# 直接源代码运行
npm run fly_pi:dev -- <参数> 就能透传
# web UI（默认端口 4695，地址 http://127.0.0.1:4695/fly_pi/）；`web` 等价于 `--mode web`
npm run fly_pi:dev web
.\fly_pi_dev.bat web
